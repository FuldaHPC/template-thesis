  % Measure the wrapped title before placing it.
  % Long titles first get a little more vertical room; only when that is not
  % enough is the font reduced.
  \newsavebox{\ThesisTitleBox}
  \newsavebox{\ThesisScaledTitleBox}
  \newlength{\ThesisTitleHeight}
  \newlength{\ThesisTitleTopSpace}
  \newlength{\ThesisTitleShift}
  \newcommand{\BuildThesisTitleBox}[2]{%
    \sbox{\ThesisTitleBox}{%
      \begin{minipage}{.88\textwidth}
        \raggedright
        {\fontsize{#1}{#2}\selectfont\bfseries\color{hpcblack}
          \ThesisTitle\par}
      \end{minipage}%
    }%
    \setlength{\ThesisTitleHeight}{%
      \dimexpr\ht\ThesisTitleBox+\dp\ThesisTitleBox\relax}%
  }

  \BuildThesisTitleBox{29}{35}
  \ifdim\ThesisTitleHeight>37mm
    \BuildThesisTitleBox{25}{30}
  \fi
  \ifdim\ThesisTitleHeight>37mm
    \BuildThesisTitleBox{22}{27}
  \fi
  \ifdim\ThesisTitleHeight>37mm
    \BuildThesisTitleBox{19}{23}
  \fi
  % Exceptionally long titles may still exceed the available height at the
  % smallest regular step.
  % Scale that final setting just enough to keep the complete title page
  % together.
  \ifdim\ThesisTitleHeight>37mm
    \sbox{\ThesisScaledTitleBox}{%
      \resizebox*{!}{37mm}{\usebox{\ThesisTitleBox}}}%
    \sbox{\ThesisTitleBox}{\usebox{\ThesisScaledTitleBox}}%
    \setlength{\ThesisTitleHeight}{%
      \dimexpr\ht\ThesisTitleBox+\dp\ThesisTitleBox\relax}%
  \fi

  % Two title lines use the original spacing.
  % For every additional amount of title height, half of that amount is
  % recovered above the central block.
  \setlength{\ThesisTitleTopSpace}{23mm}
  \ifdim\ThesisTitleHeight>70pt
    \setlength{\ThesisTitleShift}{\ThesisTitleHeight}
    \addtolength{\ThesisTitleShift}{-70pt}
    \divide\ThesisTitleShift by 2
    \addtolength{\ThesisTitleTopSpace}{-\ThesisTitleShift}
  \fi
  \ifdim\ThesisTitleTopSpace<9mm
    \setlength{\ThesisTitleTopSpace}{9mm}
  \fi
